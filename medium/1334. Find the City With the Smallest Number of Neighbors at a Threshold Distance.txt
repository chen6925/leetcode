class Solution {
public:
    int findTheCity(int n, vector<vector<int>>& edges, int distanceThreshold) 
    {   
        vector<vector<int>> dist(n,vector<int> (n,10000000));
        vector<int> distanceNum(n,0);
        int num = edges.size();
        int i,j,k;
        int answer = 0;
        for(int i = 0; i < n; i++)
        {
            dist[i][i] = 0;
        }
        for(i = 0;i < num;i++)
        {
            int u = edges[i][0];
            int v = edges[i][1];
            dist[u][v] = edges[i][2];
            dist[v][u] = edges[i][2]; 
        }
        for(i = 0;i < n;i++)
        {
            for(j = 0;j < n;j++)
            {
                for(k = 0;k < n;k++)
                {
                    dist[j][k] = min(dist[j][k],dist[j][i] + dist[i][k]);
                }
            }
        }
        
        for(i = 0;i < n;i++)
        {
            for(j = 0;j < n;j++)
            {
                if(dist[i][j] <= distanceThreshold && i != j)
                {
                    distanceNum[i]++;
                }
            }
            if(distanceNum[i] <= distanceNum[answer])
                answer = i;
        }

        return answer;
    }
};
